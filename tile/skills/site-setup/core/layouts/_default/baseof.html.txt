<script>(function () {
  var b = document.getElementById("theme-switch"), root = document.documentElement;
  if (!b) return;
  function show() {
    var dark = root.getAttribute("data-theme") === "dark";
    b.querySelector(".icon").textContent = dark ? "☀" : "☾";
    b.querySelector(".label").textContent = dark ? "Light mode" : "Dark mode";
  }
  b.hidden = false; show();
  b.addEventListener("click", function () {
    var t = root.getAttribute("data-theme") === "dark" ? "light" : "dark";
    root.setAttribute("data-theme", t);
    try { localStorage.setItem("theme", t); } catch (e) {}
    show();
  });
})();</script>
