<script>(function () {
  var tabs = document.querySelectorAll(".inpage-tabs a");
  function select(id) {
    tabs.forEach(function (t) {
      var on = t.getAttribute("aria-controls") === id;
      t.setAttribute("aria-selected", on ? "true" : "false");
      document.getElementById(t.getAttribute("aria-controls")).hidden = !on;
    });
  }
  tabs.forEach(function (t) {
    t.addEventListener("click", function (e) {
      e.preventDefault(); select(t.getAttribute("aria-controls"));
      history.replaceState(null, "", "#" + t.getAttribute("aria-controls"));
    });
  });
  select(location.hash === "#timeline" ? "timeline" : "by-week");
})();</script>{{ end }}
