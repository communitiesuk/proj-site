{{/* Coming next (ADR 0003): rows grouped by theme, each group an h3, groups
in alphabetical order with items lacking a theme last under "No theme";
oldest date first within a group, undated last. A row is the item's title
linked to its GitHub issue and, where it has one, a theme tag; it carries
data-item-theme and data-date. Where the item has a summary the row is a
<details> whose only other text is that summary and a "Discuss on GitHub"
link; otherwise it shows nothing else. An "Open all" button, rendered
hidden and only where a row expands, opens every <details>; the controls
script shows it. No text from an issue's body other than its summary.
Where there are two or more items, a controls panel is rendered with
`hidden` above the list: "Group by theme" (checked), "Order by date"
("Oldest first", "Newest first"), "Show themes" with a checked box per
theme present ("No theme" last, only where an item lacks one) and "Reset".
coming-next.js shows and wires it. */}}
{{ define "main" }}<h1>{{ .Title }}</h1>{{ with .Params.intro }}<p class="intro">{{ . }}</p>{{ end }}{{ .Content }}
{{ $repo := site.Params.repository }}{{ $items := hugo.Data.upcoming }}
{{ if $items }}
{{/* Groups: each theme in alphabetical order, then items without one. */}}
{{ $themes := slice }}{{ $none := slice }}{{ $expands := false }}
{{ range $items }}{{ if .theme }}{{ $themes = $themes | append .theme }}{{ else }}{{ $none = $none | append . }}{{ end }}{{ if .summary }}{{ $expands = true }}{{ end }}{{ end }}
{{ $groups := slice }}
{{ range sort (uniq $themes) }}{{ $t := . }}{{ $g := slice }}{{ range $items }}{{ if eq .theme $t }}{{ $g = $g | append . }}{{ end }}{{ end }}{{ $groups = $groups | append (dict "name" $t "items" $g) }}{{ end }}
{{ with $none }}{{ $groups = $groups | append (dict "name" "No theme" "items" .) }}{{ end }}
<div class="coming">
{{ if $expands }}<button type="button" class="open-all" hidden>Open all</button>{{ end }}
{{ $n := 0 }}
{{ range $groups }}
{{/* Oldest date first, undated last. */}}
{{ $dated := slice }}{{ $undated := slice }}{{ range .items }}{{ if .date }}{{ $dated = $dated | append . }}{{ else }}{{ $undated = $undated | append . }}{{ end }}{{ end }}
{{ $rows := slice }}{{ with $dated }}{{ $rows = sort . "date" }}{{ end }}{{ range $undated }}{{ $rows = $rows | append . }}{{ end }}
<section class="group" data-group="{{ .name }}"><h3>{{ .name }}</h3>
<ul class="items">{{ range $rows }}{{ $url := printf "https://github.com/%s/issues/%s" $repo (strings.TrimPrefix "#" .ref) }}<li class="row" data-item-theme="{{ .theme }}" data-date="{{ .date }}" data-index="{{ $n }}">{{ $n = add $n 1 }}{{ if .summary }}<details><summary><a href="{{ $url }}">{{ .title }}</a>{{ with .theme }} <span class="tag">{{ . }}</span>{{ end }}</summary><p>{{ .summary }}</p><p><a href="{{ $url }}">Discuss on GitHub</a></p></details>{{ else }}<a href="{{ $url }}">{{ .title }}</a>{{ with .theme }} <span class="tag">{{ . }}</span>{{ end }}{{ end }}</li>{{ end }}</ul>
</section>
{{ end }}
</div>
{{ else }}<p class="muted">Nothing queued.</p>{{ end }}{{ end }}
